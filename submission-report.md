## Design

This lab implements an IPv4 subnetting and forwarding program with three commands: info, split, and lookup. The library in lab.c handles address and route parsing, CIDR calculations, subnet splitting, and forwarding-table operations.

The CIDR calculations handle /0, /31, and /32 explicitly. A /0 uses a zero mask and covers the entire IPv4 address space. Its host count is calculated using 64-bit arithmetic to represent (2^{32}) addresses, then subtracts the network and broadcast addresses. A /31 has two usable addresses for a point-to-point link. A /32 represents one address, which is both the first and last host.

The forwarding table uses a binary trie for longest-prefix lookup. Each level represents one address bit, from most significant to least significant. Lookup follows the destination’s bits and remembers each matching route; the last match is the most specific. A default /0 route is stored at the root and is used when no more specific route matches.

## Analysis

This program performs the info, split, and lookup commands as expected. Networks are parsed correctly to provide the needed info and networks are split evenly as long as they are positive power of 2 requests. Routing lookups examine the table for longest prefix and are routed correctly. Because the example table has a default route through a gateway, every valid IPv4 address will match some route in the table. This means the program can select a route for the address; it does not guarantee that the gateway or any later router can actually deliver packets to it.

```c
bool ip_parse(const char *text, uint32_t *addr) {
  if (text == NULL || addr == NULL) {
    return false;
  }

  /* Build the host-order address one octet at a time. */
  uint32_t result = 0;
  const char *p = text;

  for (unsigned int octet = 0; octet < 4; ++octet) {
    if (*p == '\0') {
      return false;
    }

    unsigned int value = 0;
    unsigned int digits = 0;
    while (*p >= '0' && *p <= '9') {
      unsigned int digit = (unsigned int)(*p - '0');
      /* Check the limit before multiplying, so long input cannot wrap. */
      if (value > 255U / 10U || (value == 255U / 10U && digit > 255U % 10U)) {
        return false;
      }
      value = value * 10U + digit;
      ++digits;
      ++p;
    }

    /* Only one-digit zero is allowed; "0", not "00" or "010". */
    if (digits == 0 || (digits > 1 && *(p - digits) == '0')) {
      return false;
    }

    /* Each successful octet becomes the next most-significant byte. */
    result = (result << 8) | value;
    if (octet < 3) {
      if (*p != '.') {
        return false;
      }
      ++p;
      if (*p == '\0') {
        return false;
      }
    } else if (*p != '\0') {
      return false;
    }
  }

  *addr = result;
  return true;
}

void ip_format(uint32_t addr, char out[IP_STR_LEN]) {
  /* Extract bytes from most significant to least significant. */
  (void)snprintf(out, IP_STR_LEN, "%u.%u.%u.%u",
                 (unsigned int)((addr >> 24) & 0xffU),
                 (unsigned int)((addr >> 16) & 0xffU),
                 (unsigned int)((addr >> 8) & 0xffU),
                 (unsigned int)(addr & 0xffU));
}

bool cidr_parse(const char *text, cidr_t *block) {
  if (text == NULL || block == NULL) {
    return false;
  }

  const char *slash = strchr(text, '/');
  /* Exactly one slash is required. */
  if (slash == NULL || strchr(slash + 1, '/') != NULL) {
    return false;
  }

  /* Copy the address portion so ip_parse() receives a complete C string. */
  size_t address_length = (size_t)(slash - text);
  char address[IP_STR_LEN];
  if (address_length == 0 || address_length >= sizeof(address)) {
    return false;
  }
  memcpy(address, text, address_length);
  address[address_length] = '\0';

  uint32_t addr;
  if (!ip_parse(address, &addr)) {
    return false;
  }

  const char *p = slash + 1;
  if (*p == '\0') {
    return false;
  }
  unsigned int prefix = 0;
  /* /0 is valid, but multi-digit prefixes cannot have a leading zero. */
  if (*p == '0' && *(p + 1) >= '0' && *(p + 1) <= '9') {
    return false;
  }
  while (*p >= '0' && *p <= '9') {
    unsigned int digit = (unsigned int)(*p - '0');
    prefix = prefix * 10U + digit;
    ++p;
    if (prefix > 32U) {
      return false;
    }
  }
  if (p == slash + 1 || *p != '\0') {
    return false;
  }

  cidr_t parsed = {addr, (uint8_t)prefix};
  *block = parsed;
  return true;
}

void cidr_format(cidr_t block, char out[CIDR_STR_LEN]) {
  char address[IP_STR_LEN];
  ip_format(block.addr, address);
  unsigned int prefix = block.prefix;
  /* Match cidr_mask(): prefixes above 32 are represented as /32. */
  if (prefix > 32U) {
    prefix = 32U;
  }
  (void)snprintf(out, CIDR_STR_LEN, "%s/%u", address, prefix);
}

uint32_t cidr_mask(uint8_t prefix) {
  /* Avoid the undefined shift-by-32 produced by a /0 formula. */
  if (prefix == 0) {
    return 0;
  }
  /* Larger prefixes are treated as /32 throughout the API. */
  if (prefix > 32) {
    prefix = 32;
  }
  return UINT32_MAX << (32U - (uint32_t)prefix);
}

uint32_t cidr_network(cidr_t block) {
  /* Clearing with the mask retains only the network portion. */
  return block.addr & cidr_mask(block.prefix);
}

uint32_t cidr_broadcast(cidr_t block) {
  return cidr_network(block) | ~cidr_mask(block.prefix);
}

uint32_t cidr_first_host(cidr_t block) {
  uint32_t prefix = block.prefix > 32 ? 32U : block.prefix;
  uint32_t network = cidr_network(block);
  /* /31 and /32 have no excluded network address. */
  return prefix < 31 ? network + 1U : network;
}

uint32_t cidr_last_host(cidr_t block) {
  uint32_t prefix = block.prefix > 32 ? 32U : block.prefix;
  uint32_t broadcast = cidr_broadcast(block);
  /* /31 and /32 use the broadcast address as a usable endpoint. */
  return prefix < 31 ? broadcast - 1U : broadcast;
}

uint64_t cidr_host_count(cidr_t block) {
  uint32_t prefix = block.prefix > 32 ? 32U : block.prefix;
  /* Handle the point-to-point and single-host cases explicitly. */
  if (prefix == 32) {
    return 1;
  }
  if (prefix == 31) {
    return 2;
  }
  /* Compute in 64 bits because a /0 contains 2^32 addresses. */
  return (UINT64_C(1) << (32U - prefix)) - 2U;
}

bool cidr_contains(cidr_t block, uint32_t addr) {
  /* Masking both sides removes the bits that the block does not own. */
  return cidr_network(block) == (addr & cidr_mask(block.prefix));
}

bool cidr_is_network(cidr_t block) {
  /* Host bits are absent exactly when the network address is unchanged. */
  return cidr_network(block) == block.addr;
}

bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out) {
  /* A size must be both nonzero and an exact power of two. */
  if (out == NULL || !cidr_is_network(block) || n == 0 || (n & (n - 1)) != 0 || i >= n) {
    return false;
  }

  /* Find log2(n), the number of host bits borrowed from the block. */
  uint32_t borrowed_bits = 0;
  uint32_t remaining = n;
  while (remaining > 1) {
    ++borrowed_bits;
    remaining >>= 1;
  }

  uint32_t prefix = block.prefix > 32 ? 32U : block.prefix;
  if (prefix + borrowed_bits > 32) {
    return false;
  }

  uint32_t new_prefix = prefix + borrowed_bits;
  uint32_t host_bits = 32U - new_prefix;
  /* Each index advances by one equally sized subnet. */
  uint64_t subnet_size = UINT64_C(1) << host_bits;
  uint32_t subnet_address = (uint32_t)(block.addr + i * subnet_size);

  out->addr = subnet_address;
  out->prefix = (uint8_t)new_prefix;
  return true;
}

bool route_parse(const char *line, route_t *route) {
  if (line == NULL || route == NULL) {
    return false;
  }

  /* Trim outer whitespace, then keep field slices inside the original line. */
  const char *begin = line;
  while (*begin != '\0' && isspace((unsigned char)*begin)) {
    ++begin;
  }
  const char *end = line + strlen(line);
  while (end > begin && isspace((unsigned char)end[-1])) {
    --end;
  }
  if (begin == end) {
    return false;
  }

  const char *fields[3];
  size_t lengths[3];
  const char *cursor = begin;
  for (size_t i = 0; i < 3; ++i) {
    while (cursor < end && (*cursor == ' ' || *cursor == '\t')) {
      ++cursor;
    }
    if (cursor == end) {
      return false;
    }

    fields[i] = cursor;
    while (cursor < end && *cursor != ' ' && *cursor != '\t') {
      if (isspace((unsigned char)*cursor)) {
        return false;
      }
      ++cursor;
    }
    lengths[i] = (size_t)(cursor - fields[i]);
  }

  if (cursor != end) {
    return false;
  }

  route_t parsed = {0};
  /* Parse into a temporary so invalid input never partially changes route. */
  if (lengths[0] == sizeof("default") - 1 &&
      memcmp(fields[0], "default", sizeof("default") - 1) == 0) {
    parsed.dest = (cidr_t){0, 0};
  } else {
    char destination[CIDR_STR_LEN];
    if (lengths[0] >= sizeof(destination)) {
      return false;
    }
    memcpy(destination, fields[0], lengths[0]);
    destination[lengths[0]] = '\0';
    if (!cidr_parse(destination, &parsed.dest) ||
        !cidr_is_network(parsed.dest)) {
      return false;
    }
  }

  if (lengths[1] == sizeof("direct") - 1 &&
      memcmp(fields[1], "direct", sizeof("direct") - 1) == 0) {
    /* Directly connected routes have no next-hop address. */
    parsed.gateway = 0;
  } else {
    char gateway[IP_STR_LEN];
    if (lengths[1] >= sizeof(gateway)) {
      return false;
    }
    memcpy(gateway, fields[1], lengths[1]);
    gateway[lengths[1]] = '\0';
    if (!ip_parse(gateway, &parsed.gateway)) {
      return false;
    }
  }

  if (lengths[2] == 0 || lengths[2] >= sizeof(parsed.dev)) {
    return false;
  }
  memcpy(parsed.dev, fields[2], lengths[2]);
  parsed.dev[lengths[2]] = '\0';
  *route = parsed;
  return true;
}

bool table_add(fwd_table_t *table, const route_t *route) {
  /* Only canonical networks with a usable, terminated device name are valid. */
  if (table == NULL || route == NULL || route->dest.prefix > 32 ||
      !cidr_is_network(route->dest)) {
    return false;
  }

  size_t dev_length = 0;
  while (dev_length < sizeof(route->dev) && route->dev[dev_length] != '\0') {
    ++dev_length;
  }
  if (dev_length == 0 || dev_length == sizeof(route->dev)) {
    return false;
  }

  if (table->count >= TRIE_INDEX_NONE) {
    return false;
  }

  /* Walk the existing prefix path to detect duplicates and find missing nodes. */
  uint32_t node_index = 0;
  uint32_t depth = 0;
  while (depth < route->dest.prefix) {
    uint32_t bit = (route->dest.addr >> (31U - depth)) & 1U;
    uint32_t child = table->trie[node_index].child[bit];
    if (child == TRIE_INDEX_NONE) {
      break;
    }
    node_index = child;
    ++depth;
  }
  if (depth == route->dest.prefix &&
      table->trie[node_index].route_index != TRIE_INDEX_NONE) {
    return false;
  }

  size_t missing_nodes = (size_t)route->dest.prefix - depth;
  if (missing_nodes > (size_t)TRIE_INDEX_NONE - table->trie_count) {
    return false;
  }
  size_t needed_nodes = table->trie_count + missing_nodes;
  if (needed_nodes > table->trie_capacity) {
    size_t new_capacity = table->trie_capacity;
    while (new_capacity < needed_nodes) {
      // Reaching this capacity requires impractically large trie storage.
      // GCOVR_EXCL_START
      if (new_capacity > (size_t)TRIE_INDEX_NONE / 2) {
        new_capacity = (size_t)TRIE_INDEX_NONE;
        break;
      }
      // GCOVR_EXCL_STOP
      new_capacity *= 2;
    } // GCOVR_EXCL_LINE
    // These overflow guards are unreachable with the table's 32-bit index
    // limits on this 64-bit build.
    // GCOVR_EXCL_START
    if (new_capacity < needed_nodes ||
        new_capacity > SIZE_MAX / sizeof(*table->trie)) {
      return false;
    }
    // GCOVR_EXCL_STOP
    trie_node_t *new_trie =
        realloc(table->trie, new_capacity * sizeof(*table->trie));
    // GCOVR_EXCL_START
    if (new_trie == NULL) {
      return false;
    }
    // GCOVR_EXCL_STOP
    table->trie = new_trie;
    table->trie_capacity = new_capacity;
  } // GCOVR_EXCL_LINE

  if (table->count == table->capacity) {
    /* Grow geometrically to keep repeated additions efficient. */
    size_t new_capacity = table->capacity == 0 ? 8 : table->capacity * 2;
    // The route-count limit prevents this size overflow on this 64-bit build.
    // GCOVR_EXCL_START
    if (new_capacity < table->capacity ||
        new_capacity > SIZE_MAX / sizeof(*table->routes)) {
      return false;
    }
    // GCOVR_EXCL_STOP
    route_t *new_routes =
        realloc(table->routes, new_capacity * sizeof(*table->routes));
    // GCOVR_EXCL_START
    if (new_routes == NULL) {
      return false;
    }
    // GCOVR_EXCL_STOP
    table->routes = new_routes;
    table->capacity = new_capacity;
  }

  /* Create any missing path nodes, then store this route at its prefix node. */
  node_index = 0;
  for (depth = 0; depth < route->dest.prefix; ++depth) {
    uint32_t bit = (route->dest.addr >> (31U - depth)) & 1U;
    uint32_t child = table->trie[node_index].child[bit];
    if (child == TRIE_INDEX_NONE) {
      child = (uint32_t)table->trie_count++;
      table->trie[child] = (trie_node_t){{TRIE_INDEX_NONE, TRIE_INDEX_NONE},
                                         TRIE_INDEX_NONE};
      table->trie[node_index].child[bit] = child;
    }
    node_index = child;
  }

  table->routes[table->count] = *route;
  table->trie[node_index].route_index = (uint32_t)table->count;
  ++table->count;
  return true;
}

bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line) {
  if (bad_line != NULL) {
    *bad_line = 0;
  }
  if (table == NULL || in == NULL) {
    return false;
  }

  char *line = NULL;
  size_t capacity = 0;
  size_t line_number = 0;
  ssize_t length;
  /* getline handles arbitrarily long input lines without a fixed-size limit. */
  while ((length = getline(&line, &capacity, in)) != -1) {
    ++line_number;
    if (memchr(line, '\0', (size_t)length) != NULL) {
      if (bad_line != NULL) {
        *bad_line = line_number;
      }
      free(line);
      return false;
    }
    char *comment = memchr(line, '#', (size_t)length);
    if (comment != NULL) {
      /* Strip comments before deciding whether this line contains a route. */
      *comment = '\0';
    }

    const char *content = line;
    while (*content != '\0' && isspace((unsigned char)*content)) {
      ++content;
    }
    if (*content == '\0') {
      continue;
    }

    route_t route;
    /* Keep successfully added earlier routes if this line is invalid. */
    if (!route_parse(line, &route) || !table_add(table, &route)) {
      if (bad_line != NULL) {
        *bad_line = line_number;
      }
      free(line);
      return false;
    }
  }

  free(line);
  /* A read error is distinct from a clean end of file. */
  return feof(in) && !ferror(in);
}

const route_t *table_lookup(const fwd_table_t *table, uint32_t dest) {
  if (table == NULL) {
    return NULL;
  }

  /* Remember each matching route while descending at most 32 address bits. */
  uint32_t route_index = table->trie[0].route_index;
  uint32_t node_index = 0;
  for (uint32_t depth = 0; depth < 32; ++depth) {
    uint32_t bit = (dest >> (31U - depth)) & 1U;
    uint32_t child = table->trie[node_index].child[bit];
    if (child == TRIE_INDEX_NONE) {
      break;
    }
    node_index = child;
    if (table->trie[node_index].route_index != TRIE_INDEX_NONE) {
      route_index = table->trie[node_index].route_index;
    }
  }
  return route_index == TRIE_INDEX_NONE ? NULL : &table->routes[route_index];
}
```

```c

#ifndef LAB_H
#define LAB_H

#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>

/*
 * Every address in this file is a uint32_t in host byte order, so the first
 * octet of 192.168.1.10 is the most significant byte: 0xC0A8010A.
 */

/** Room for the longest dotted quad, "255.255.255.255", and its '\0'. */
#define IP_STR_LEN 16

/** Room for the longest CIDR string, "255.255.255.255/32", and its '\0'. */
#define CIDR_STR_LEN 19

/** Room for an interface name; IFNAMSIZ on Linux, including the '\0'. */
#define DEV_NAME_LEN 16

/** An address block in CIDR notation: an address and a prefix length. */
typedef struct {
  uint32_t addr;  /**< any address in the block, host byte order */
  uint8_t prefix; /**< 0 to 32 */
} cidr_t;

/** One entry in a forwarding table. */
typedef struct {
  cidr_t dest;            /**< destination block; host bits must be zero */
  uint32_t gateway;       /**< next hop, or 0 when the block is directly connected */
  char dev[DEV_NAME_LEN]; /**< outgoing interface, for example "eth0" */
} route_t;

/** A forwarding table. Its layout is private to lab.c. */
typedef struct fwd_table fwd_table_t;

/* ------------------------------------------------------------------------
 * Task 1 - Address parsing and formatting
 * --------------------------------------------------------------------- */

/**
 * @brief Parse a dotted quad such as "192.168.1.10".
 *
 * Exactly four decimal octets from 0 to 255 separated by single periods, with
 * nothing before, between or after them. An octet with a leading zero ("010")
 * is rejected, because inet_aton() reads it as octal and that is never what a
 * person typing it meant.
 *
 * @param text The string to parse.
 * @param addr Where the address is stored on success; untouched on failure.
 * @return true if text is a valid dotted quad, false otherwise (including NULL).
 */
bool ip_parse(const char *text, uint32_t *addr);

/**
 * @brief Format an address as a dotted quad.
 * @param addr The address, host byte order.
 * @param out A buffer of at least IP_STR_LEN bytes.
 */
void ip_format(uint32_t addr, char out[IP_STR_LEN]);

/**
 * @brief Parse a block in CIDR notation such as "192.168.1.0/24".
 *
 * The address follows the rules of ip_parse() and the prefix is a decimal
 * number from 0 to 32 with no leading zero. The slash and the prefix are
 * required. Host bits may be set ("132.178.227.11/25" is an interface
 * address), so use cidr_is_network() when they must not be.
 *
 * @param text The string to parse.
 * @param block Where the block is stored on success; untouched on failure.
 * @return true if text is valid CIDR notation, false otherwise (including NULL).
 */
bool cidr_parse(const char *text, cidr_t *block);

/**
 * @brief Format a block as "a.b.c.d/p", using the address exactly as stored.
 *
 * A prefix over 32 is written as 32, the way cidr_mask() treats it.
 *
 * @param block The block.
 * @param out A buffer of at least CIDR_STR_LEN bytes.
 */
void cidr_format(cidr_t block, char out[CIDR_STR_LEN]);

/* ------------------------------------------------------------------------
 * Task 2 - Subnet arithmetic
 * --------------------------------------------------------------------- */

/**
 * @brief The subnet mask for a prefix length: /24 is 255.255.255.0.
 *
 * Careful with /0. Shifting a 32 bit value by 32 is undefined behavior in C.
 *
 * @param prefix 0 to 32; anything larger is treated as 32.
 * @return The mask, host byte order.
 */
uint32_t cidr_mask(uint8_t prefix);

/** @brief The lowest address in the block: the address with the host bits cleared. */
uint32_t cidr_network(cidr_t block);

/** @brief The highest address in the block: the address with the host bits set. */
uint32_t cidr_broadcast(cidr_t block);

/**
 * @brief The first address that can be given to a host.
 *
 * For /0 to /30 that is the network address plus one. A /31 is a point to
 * point link (RFC 3021) where both addresses are usable, and a /32 is a single
 * host, so for those two the first host is the network address.
 */
uint32_t cidr_first_host(cidr_t block);

/**
 * @brief The last address that can be given to a host.
 *
 * The broadcast address minus one for /0 to /30, and the broadcast address
 * itself for /31 and /32 (see cidr_first_host()).
 */
uint32_t cidr_last_host(cidr_t block);

/**
 * @brief How many addresses in the block can be given to hosts.
 *
 * 2^(32 - prefix) - 2 for /0 to /30, 2 for /31, and 1 for /32. For a /0 the
 * 2^32 in that formula needs 33 bits, more than a uint32_t holds, which is why
 * this returns a uint64_t.
 */
uint64_t cidr_host_count(cidr_t block);

/** @brief true if addr falls inside the block. */
bool cidr_contains(cidr_t block, uint32_t addr);

/** @brief true if the block's host bits are all zero, so it names a network. */
bool cidr_is_network(cidr_t block);

/**
 * @brief The i-th of n equally sized subnets of a block.
 *
 * Splitting a /24 into 4 gives four /26 blocks, and subnet 0 is the lowest.
 * Equal sizes are only possible when n is a power of two, and the new prefix
 * can not pass 32.
 *
 * @param block The block to split; it must be a network (cidr_is_network()).
 * @param n How many subnets, a power of two from 1 up.
 * @param i Which subnet, 0 to n - 1.
 * @param out Where the subnet is stored on success; untouched on failure.
 * @return false if out is NULL, the block is not a network, n is not a power
 *         of two, the split would need a prefix longer than 32, or i is out of
 *         range.
 */
bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out);

/* ------------------------------------------------------------------------
 * Task 3 - Longest prefix match
 * --------------------------------------------------------------------- */

/**
 * @brief Parse one line of a forwarding table file.
 *
 * A line is three fields separated by spaces or tabs:
 *
 *     <destination> <gateway> <device>
 *
 * The destination is a block in CIDR notation with no host bits set, or the
 * word "default" for 0.0.0.0/0. The gateway is a dotted quad, or the word
 * "direct" for a directly connected block (stored as 0). The device is 1 to
 * DEV_NAME_LEN - 1 characters. Leading and trailing whitespace, including a
 * trailing newline, is ignored. Comments are not handled here.
 *
 * @param line The line.
 * @param route Where the route is stored on success; untouched on failure.
 * @return true if the line is a valid route, false otherwise (including NULL).
 */
bool route_parse(const char *line, route_t *route);

/**
 * @brief Create an empty forwarding table.
 * @return The table, which the caller frees with table_destroy(), or NULL if
 *         memory could not be allocated.
 */
fwd_table_t *table_create(void);

/** @brief Free a table and everything in it. NULL is allowed and does nothing. */
void table_destroy(fwd_table_t *table);

/** @brief The number of routes in the table; 0 for NULL. */
size_t table_count(const fwd_table_t *table);

#ifdef TEST
/** @brief Set the route count for exercising capacity guards in tests. */
void table_test_set_count(fwd_table_t *table, size_t count);
/** @brief Set the trie-node count for exercising capacity guards in tests. */
void table_test_set_trie_count(fwd_table_t *table, size_t count);
#endif

/**
 * @brief Add a copy of a route to the table, growing it as needed.
 * @return false if either argument is NULL, the destination has host bits
 *         set, the table already holds a route to the same block, or memory
 *         could not be allocated. The table is unchanged on failure.
 */
bool table_add(fwd_table_t *table, const route_t *route);

/**
 * @brief Read a forwarding table file and add every route in it.
 *
 * Everything from a '#' to the end of a line is a comment, and a line that is
 * blank once the comment is removed is skipped. Every other line must be a
 * valid route (route_parse()) that table_add() accepts.
 *
 * @param table The table to add to.
 * @param in The open file to read.
 * @param bad_line If not NULL, set to the 1-based number of the first line
 *        that failed, or 0 if the arguments were NULL or the file could not
 *        be read.
 * @return true if every line loaded. On false, routes from the lines before
 *         the bad one stay in the table.
 */
bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line);

/**
 * @brief Longest prefix match: the route whose block contains dest and has the
 *        longest prefix.
 *
 * A default route (0.0.0.0/0) matches everything, so it is chosen only when
 * nothing more specific does. The order the routes were added in does not
 * matter.
 *
 * @return The matching route, which stays owned by the table, or NULL if no
 *         route matches (or table is NULL).
 */
const route_t *table_lookup(const fwd_table_t *table, uint32_t dest);

#endif // LAB_H
```

```c
#include "lab.h"
#include <errno.h>
#include <inttypes.h>

#include <stdlib.h>
#include <string.h>

#ifdef TEST
#define main main_exclude

/* Keep the usage text identical for help output and command-line errors. */
static void print_usage(FILE *out) {
  (void)fprintf(out,
                "Usage: myapp info <address>/<prefix>\n"
                "       myapp split <network>/<prefix> <count>\n"
                "       myapp lookup <table-file> <address>...\n");
}

static int report_bad_command(const char *message) {
  (void)fprintf(stderr, "myapp: %s\n", message);
  print_usage(stderr);
  return 1;
}

static int command_info(const char *text) {
  cidr_t block;
  if (!cidr_parse(text, &block)) {
    (void)fprintf(stderr, "myapp: '%s' is not a valid address/prefix\n", text);
    return 1;
  }

  char address[IP_STR_LEN];
  char netmask[IP_STR_LEN];
  char network[IP_STR_LEN];
  char broadcast[IP_STR_LEN];
  char first_host[IP_STR_LEN];
  char last_host[IP_STR_LEN];
  char prefix[5];

  /* Derive each displayed value from the parsed address and prefix. */
  ip_format(block.addr, address);
  ip_format(cidr_mask(block.prefix), netmask);
  ip_format(cidr_network(block), network);
  ip_format(cidr_broadcast(block), broadcast);
  ip_format(cidr_first_host(block), first_host);
  ip_format(cidr_last_host(block), last_host);
  (void)snprintf(prefix, sizeof(prefix), "/%u", (unsigned int)block.prefix);

  (void)printf("%-10s %s\n", "address", address);
  (void)printf("%-10s %s\n", "prefix", prefix);
  (void)printf("%-10s %s\n", "netmask", netmask);
  (void)printf("%-10s %s\n", "network", network);
  (void)printf("%-10s %s\n", "broadcast", broadcast);
  (void)printf("%-10s %s\n", "first host", first_host);
  (void)printf("%-10s %s\n", "last host", last_host);
  (void)printf("%-10s %" PRIu64 "\n", "hosts", cidr_host_count(block));
  return 0;
}

static bool parse_count(const char *text, uint32_t *count) {
  if (text == NULL || *text == '\0') {
    return false;
  }
  for (const char *p = text; *p != '\0'; ++p) {
    if (*p < '0' || *p > '9') {
      return false;
    }
  }

  /* Check conversion overflow as well as rejecting trailing characters. */
  errno = 0;
  char *end = NULL;
  unsigned long long value = strtoull(text, &end, 10);
  if (errno == ERANGE || end == text || *end != '\0' ||
      value > UINT32_MAX) {
    return false;
  }
  *count = (uint32_t)value;
  return true;
}

static int command_split(const char *text, const char *count_text) {
  cidr_t block;
  uint32_t count;
  if (!cidr_parse(text, &block)) {
    (void)fprintf(stderr, "myapp: '%s' is not a valid network/prefix\n", text);
    return 1;
  }
  if (!cidr_is_network(block)) {
    (void)fprintf(stderr, "myapp: '%s' has host bits set; expected a network\n",
                  text);
    return 1;
  }
  if (!parse_count(count_text, &count) || count == 0 ||
      (count & (count - 1U)) != 0) {
    (void)fprintf(stderr, "myapp: '%s' is not a positive power-of-two count\n",
                  count_text);
    return 1;
  }

  cidr_t subnet;
  /* Validate that this count can be represented by equal subnets up front. */
  if (!cidr_subnet(block, count, 0, &subnet)) {
    (void)fprintf(stderr,
                  "myapp: cannot split %s into %" PRIu32 " subnets\n",
                  text, count);
    return 1;
  }

  /* cidr_subnet orders results from the lowest subnet address upward. */
  for (uint32_t i = 0; i < count; ++i) {
    (void)cidr_subnet(block, count, i, &subnet);
    char formatted[CIDR_STR_LEN];
    cidr_format(subnet, formatted);
    (void)puts(formatted);
  }
  return 0;
}

static int command_lookup(const char *path, int address_count,
                          char *const addresses[]) {
  FILE *input = fopen(path, "r");
  if (input == NULL) {
    (void)fprintf(stderr, "myapp: cannot open '%s': %s\n", path,
                  strerror(errno));
    return 2;
  }

  fwd_table_t *table = table_create();
  if (table == NULL) {
    (void)fprintf(stderr, "myapp: cannot allocate forwarding table\n");
    (void)fclose(input);
    return 2;
  }

  size_t bad_line = 0;
  /* Do not begin lookups if the forwarding table is incomplete or invalid. */
  if (!table_load(table, input, &bad_line)) {
    if (bad_line != 0) {
      (void)fprintf(stderr, "myapp: %s:%zu: not a valid route\n", path,
                    bad_line);
    } else {
      (void)fprintf(stderr, "myapp: failed reading table '%s'\n", path);
    }
    table_destroy(table);
    (void)fclose(input);
    return 2;
  }
  (void)fclose(input);

  bool malformed = false;
  bool unreachable = false;
  /* Process every address so one bad input does not hide the other results. */
  for (int i = 0; i < address_count; ++i) {
    uint32_t address;
    if (!ip_parse(addresses[i], &address)) {
      (void)fprintf(stderr, "myapp: '%s' is not a valid address\n",
                    addresses[i]);
      malformed = true;
      continue;
    }

    const route_t *route = table_lookup(table, address);
    if (route == NULL) {
      (void)printf("%s unreachable\n", addresses[i]);
      unreachable = true;
      continue;
    }

    (void)printf("%s", addresses[i]);
    if (route->gateway != 0) {
      char gateway[IP_STR_LEN];
      ip_format(route->gateway, gateway);
      (void)printf(" via %s", gateway);
    }
    (void)printf(" dev %s\n", route->dev);
  }

  table_destroy(table);
  /* A malformed address takes precedence over an otherwise valid no-route. */
  if (malformed) {
    return 1;
  }
  return unreachable ? 3 : 0;
}

int main(int argc, char **argv) {
  /* No-argument invocation is a successful way to display help. */
  if (argc == 1) {
    print_usage(stdout);
    return 0;
  }
  /* Dispatch only after checking each command's exact argument count. */
  if (strcmp(argv[1], "info") == 0) {
    if (argc != 3) {
      return report_bad_command("info requires exactly one address/prefix");
    }
    return command_info(argv[2]);
  }
  if (strcmp(argv[1], "split") == 0) {
    if (argc != 4) {
      return report_bad_command("split requires a network/prefix and count");
    }
    return command_split(argv[2], argv[3]);
  }
  if (strcmp(argv[1], "lookup") == 0) {
    if (argc < 4) {
      return report_bad_command("lookup requires a table file and addresses");
    }
    return command_lookup(argv[2], argc - 3, &argv[3]);
  }
  return report_bad_command("unknown command");
}
```

```c
void test_ip_parse_boundaries_and_malformed_inputs(void) {
```
